    48→    48→function titleCase(str: string): string {
    49→    49→  return str
    50→    50→    .split(/\s+/)
    51→    51→    .map(word => word.charAt(0).toUpperCase() + word.slice(1).toLowerCase())
    52→    52→    .join(' ')
    53→    53→}

   116→   116→
   117→   117→function ListSelectionView() {
   118→   118→  const { token, lists, setLists, setActiveList } = useAppStore()
   119→   119→  const [loading, setLoading] = useState(true)
   120→   120→  const [dialogOpen, setDialogOpen] = useState(false)
   121→   121→  const [newListName, setNewListName] = useState('')
   122→   122→  const [creating, setCreating] = useState(false)
   123→   123→  const [deletingListId, setDeletingListId] = useState<string | null>(null)
   124→   124→  const [checkingAllListId, setCheckingAllListId] = useState<string | null>(null)
   125→   125→  const [editDialogOpen, setEditDialogOpen] = useState(false)
   126→   126→  const [editListId, setEditListId] = useState<string | null>(null)
   127→   127→  const [editListName, setEditListName] = useState('')
   128→   128→  const [editing, setEditing] = useState(false)
   129→   129→
   130→   130→  const fetchLists = useCallback(async () => {
   131→   131→    if (!token) return
   132→   132→    setLoading(true)
   133→   133→    try {
   134→   134→      const res = await fetch('/api/lists', { headers: authHeaders(token) })
   135→   135→      if (res.ok) {
   136→   136→        const data = await res.json()
   137→   137→        const mapped: ShoppingList[] = (data.lists ?? []).map(
   138→   138→          (l: {
   139→   139→            id: string
   140→   140→            name: string
   141→   141→            createdBy: string
   142→   142→            familyId?: string
   143→   143→            createdAt: string
   144→   144→            totalCount: number
   145→   145→            completedCount: number
   146→   146→          }) => ({
   147→   147→            id: l.id,
   148→   148→            name: l.name,
   149→   149→            createdBy: l.createdBy,
   150→   150→            familyId: l.familyId ?? '',
   151→   151→            createdAt: l.createdAt,
   152→   152→            _count: { items: l.totalCount, completed: l.completedCount },
   153→   153→          }),
   154→   154→        )
   155→   155→        setLists(mapped)
   156→   156→      }
   157→   157→    } catch {
   158→   158→      // silent
   159→   159→    } finally {
   160→   160→      setLoading(false)
   161→   161→    }
   162→   162→  }, [token, setLists])
   163→   163→
   164→   164→  useEffect(() => {
   165→   165→    fetchLists()
   166→   166→  }, [fetchLists])
   167→   167→
   168→   168→  const handleCreate = async () => {
   169→   169→    if (!token || !newListName.trim()) return
   170→   170→    setCreating(true)
   171→   171→    try {
   172→   172→      const res = await fetch('/api/lists', {
   173→   173→        method: 'POST',
   174→   174→        headers: { 'Content-Type': 'application/json', ...authHeaders(token) },
   175→   175→        body: JSON.stringify({ name: newListName.trim() }),
   176→   176→      })
   177→   177→      if (res.ok) {
   178→   178→        toast.success('Liste oluşturuldu!')
   179→   179→        setNewListName('')
   180→   180→        setDialogOpen(false)
   181→   181→        fetchLists()
   182→   182→      } else {
   183→   183→        toast.error('Liste oluşturulamadı')
   184→   184→      }
   185→   185→    } catch {
   186→   186→      toast.error('Bağlantı hatası')
   187→   187→    } finally {
   188→   188→      setCreating(false)
   189→   189→    }
   190→   190→  }
   191→   191→
   192→   192→  const handleDeleteList = async (listId: string, e: React.MouseEvent) => {
   193→   193→    e.stopPropagation()
   194→   194→    if (!token) return
   195→   195→    setDeletingListId(listId)
   196→   196→    try {
   197→   197→      const res = await fetch(`/api/lists/${listId}`, {
   198→   198→        method: 'DELETE',
   199→   199→        headers: authHeaders(token),
   200→   200→      })
   201→   201→      if (res.ok) {
   202→   202→        toast.success('Liste silindi')
   203→   203→        fetchLists()
   204→   204→      } else {
   205→   205→        toast.error('Liste silinemedi')
   206→   206→      }
   207→   207→    } catch {
   208→   208→      toast.error('Bağlantı hatası')
   209→   209→    } finally {
   210→   210→      setDeletingListId(null)
   211→   211→    }
   212→   212→  }
   213→   213→
   214→   214→  const handleCheckAll = async (list: ShoppingList, e: React.MouseEvent) => {
   215→   215→    e.stopPropagation()
   216→   216→    if (!token) return
   217→   217→    const total = list._count?.items ?? 0
   218→   218→    const completed = list._count?.completed ?? 0
   219→   219→    if (total === 0 || total === completed) return
   220→   220→
   221→   221→    setCheckingAllListId(list.id)
   222→   222→    try {
   223→   223→      // Fetch all items for the list
   224→   224→      const itemsRes = await fetch(`/api/lists/items?listId=${list.id}`, {
   225→   225→        headers: authHeaders(token),
   226→   226→      })
   227→   227→      if (!itemsRes.ok) {
   228→   228→        toast.error('Ürünler yüklenemedi')
   229→   229→        return
   230→   230→      }
   231→   231→      const itemsData = await itemsRes.json()
   232→   232→      const incompleteItems = (itemsData.items ?? []).filter((i: ShoppingItem) => !i.completed)
   233→   233→
   234→   234→      // Toggle each incomplete item
   235→   235→      for (const item of incompleteItems) {
   236→   236→        await fetch('/api/lists/items', {
   237→   237→          method: 'PATCH',
   238→   238→          headers: { 'Content-Type': 'application/json', ...authHeaders(token) },
   239→   239→          body: JSON.stringify({ itemId: item.id }),
   240→   240→        })
   241→   241→      }
   242→   242→
   243→   243→      toast.success('Tüm ürünler tamamlandı olarak işaretlendi')
   244→   244→      fetchLists()
   245→   245→    } catch {
   246→   246→      toast.error('Bağlantı hatası')
   247→   247→    } finally {
   248→   248→      setCheckingAllListId(null)
   249→   249→    }
   250→   250→  }
   251→   251→
   252→   252→  const handleOpenList = (list: ShoppingList) => {
   253→   253→    setActiveList(list)
   254→   254→  }
   255→   255→
   256→   256→  const openEditDialog = (list: ShoppingList, e: React.MouseEvent) => {
   257→   257→    e.stopPropagation()
   258→   258→    setEditListId(list.id)
   259→   259→    setEditListName(list.name)
   260→   260→    setEditDialogOpen(true)
   261→   261→  }
   262→   262→
   263→   263→  const handleEditList = async () => {
   264→   264→    if (!token || !editListId || !editListName.trim()) return
   265→   265→    setEditing(true)
   266→   266→    try {
   267→   267→      const res = await fetch(`/api/lists/${editListId}`, {
   268→   268→        method: 'PATCH',
   269→   269→        headers: { 'Content-Type': 'application/json', ...authHeaders(token) },
   270→   270→        body: JSON.stringify({ name: editListName.trim() }),
   271→   271→      })
   272→   272→      if (res.ok) {
   273→   273→        toast.success('Liste güncellendi')
   274→   274→        setEditDialogOpen(false)
   275→   275→        fetchLists()
   276→   276→      } else {
   277→   277→        toast.error('Liste güncellenemedi')
   278→   278→      }
   279→   279→    } catch {
   280→   280→      toast.error('Bağlantı hatası')
   281→   281→    } finally {
   282→   282→      setEditing(false)
   283→   283→    }
   284→   284→  }
   285→   285→
   286→   286→  if (loading) {
   287→   287→    return (
   288→   288→      <div className="flex min-h-screen items-center justify-center bg-background">
   289→   289→        <Loader2 className="size-8 animate-spin text-primary" />
   290→   290→      </div>
   291→   291→    )
   292→   292→  }
   293→   293→
   294→   294→  return (
   295→   295→    <main className="mx-auto min-h-screen w-full max-w-lg bg-background px-4 pb-24 pt-6">
   296→   296→      <motion.div
   297→   297→        variants={containerVariants}
   298→   298→        initial="hidden"
   299→   299→        animate="visible"
   300→   300→        className="flex flex-col gap-6"
   301→   301→      >
   302→   302→        {/* ─── Header ─── */}
   303→   303→        <motion.section variants={itemVariants} className="flex items-center justify-between">
   304→   304→          <div className="flex items-center gap-3">
   305→   305→            <div className="flex size-10 items-center justify-center rounded-xl bg-primary/15">
   306→   306→              <ShoppingBag className="size-5 text-primary" />
   307→   307→            </div>
   308→   308→            <h1 className="text-2xl font-bold text-primary">Alışveriş Listeleri</h1>
   309→   309→          </div>
   310→   310→          <Button
   311→   311→            onClick={() => setDialogOpen(true)}
   312→   312→            className="flex size-10 items-center justify-center rounded-full bg-primary text-primary-foreground hover:bg-primary/90"
   313→   313→            size="icon"
   314→   314→          >
   315→   315→            <Plus className="size-5" />
   316→   316→          </Button>
   317→   317→        </motion.section>
   318→   318→
   319→   319→        {/* ─── Lists ─── */}
   320→   320→        {lists.length === 0 ? (
   321→   321→          <motion.div
   322→   322→            initial={{ opacity: 0, scale: 0.95 }}
   323→   323→            animate={{ opacity: 1, scale: 1 }}
   324→   324→            transition={{ duration: 0.4, delay: 0.2 }}
   325→   325→            className="glass-card flex flex-col items-center justify-center gap-3 rounded-2xl px-6 py-16"
   326→   326→          >
   327→   327→            <div className="flex size-16 items-center justify-center rounded-full bg-primary/5">
   328→   328→              <ShoppingCart className="size-8 text-muted-foreground/30" />
   329→   329→            </div>
   330→   330→            <div className="flex flex-col items-center gap-1.5">
   331→   331→              <p className="text-sm font-medium text-muted-foreground">
   332→   332→                Henüz liste oluşturulmadı
   333→   333→              </p>
   334→   334→              <p className="text-xs text-muted-foreground/50">
   335→   335→                Yeni bir alışveriş listesi oluşturmak için + butonuna tıklayın
   336→   336→              </p>
   337→   337→            </div>
   338→   338→          </motion.div>
   339→   339→        ) : (
   340→   340→          <motion.div variants={containerVariants} className="flex flex-col gap-3">
   341→   341→            {lists.map((list) => {
   342→   342→              const total = list._count?.items ?? 0
   343→   343→              const completed = list._count?.completed ?? 0
   344→   344→              const progress = total > 0 ? (completed / total) * 100 : 0
   345→   345→              const isComplete = total > 0 && completed === total
   346→   346→
   347→   347→              return (
   348→   348→                <motion.div
   349→   349→                  key={list.id}
   350→   350→                  variants={itemVariants}
   351→   351→                  whileHover={cardHover}
   352→   352→                  whileTap={{ scale: 0.98 }}
   353→   353→                  onClick={() => handleOpenList(list)}
   354→   354→                  className="glass-card hover-glow cursor-pointer rounded-2xl p-4"
   355→   355→                >
   356→   356→                  <div className="flex items-start justify-between gap-3">
   357→   357→                    <div className="flex items-start gap-3">
   358→   358→                      <div
   359→   359→                        className={`flex size-10 shrink-0 items-center justify-center rounded-xl ${isComplete ? 'bg-green-500/15' : 'bg-primary/15'}`}
   360→   360→                      >
   361→   361→                        {isComplete ? (
   362→   362→                          <Check className="size-5 text-green-500" />
   363→   363→                        ) : (
   364→   364→                          <Package className="size-5 text-primary" />
   365→   365→                        )}
   366→   366→                      </div>
   367→   367→                      <div className="flex flex-col gap-1">
   368→   368→                        <h3 className="text-sm font-semibold text-foreground">
   369→   369→                          {list.name}
   370→   370→                        </h3>
   371→   371→                        <p className="text-xs text-muted-foreground">
   372→   372→                          <span className="font-medium text-primary">
   373→   373→                            {completed}
   374→   374→                          </span>{' '}
   375→   375→                          / {total} tamamlandı
   376→   376→                        </p>
   377→   377→                      </div>
   378→   378→                    </div>
   379→   379→                    <div className="flex shrink-0 items-center gap-0.5">
   380→   380→                      <Button
   381→   381→                        variant="ghost"
   382→   382→                        size="icon"
   383→   383→                        className="size-7 text-primary/60 hover:bg-primary/10 hover:text-primary"
   384→   384→                        onClick={(e) => openEditDialog(list, e)}
   385→   385→                      >
   386→   386→                        <Pencil className="size-3.5" />
   387→   387→                      </Button>
   388→   388→                      {!isComplete && total > 0 && (
   389→   389→                        <Button
   390→   390→                          variant="ghost"
   391→   391→                          size="icon"
   392→   392→                          className="size-7 text-amber-500/70 hover:bg-amber-500/10 hover:text-amber-500"
   393→   393→                          onClick={(e) => handleCheckAll(list, e)}
   394→   394→                          disabled={checkingAllListId === list.id}
   395→   395→                        >
   396→   396→                          {checkingAllListId === list.id ? (
   397→   397→                            <Loader2 className="size-3.5 animate-spin" />
   398→   398→                          ) : (
   399→   399→                            <CheckCircle className="size-3.5" />
   400→   400→                          )}
   401→   401→                        </Button>
   402→   402→                      )}
   403→   403→                      <Button
   404→   404→                        variant="ghost"
   405→   405→                        size="icon"
   406→   406→                        className="size-7 text-destructive/70 hover:bg-destructive/10 hover:text-destructive"
   407→   407→                        onClick={(e) => handleDeleteList(list.id, e)}
   408→   408→                        disabled={deletingListId === list.id}
   409→   409→                      >
   410→   410→                        {deletingListId === list.id ? (
   411→   411→                          <Loader2 className="size-3.5 animate-spin" />
   412→   412→                        ) : (
   413→   413→                          <Trash2 className="size-3.5" />
   414→   414→                        )}
   415→   415→                      </Button>
   416→   416→                      {isComplete && (
   417→   417→                        <Badge
   418→   418→                          className="shrink-0 bg-green-500/15 text-xs text-green-500 hover:bg-green-500/20"
   419→   419→                        >
   420→   420→                          <Check className="size-3 mr-1" />
   421→   421→                          Tamamlandı
   422→   422→                        </Badge>
   423→   423→                      )}
   424→   424→                    </div>
   425→   425→                  </div>
   426→   426→
   427→   427→                  {/* Progress bar */}
   428→   428→                  <div className="mt-3 h-1.5 w-full overflow-hidden rounded-full bg-border">
   429→   429→                    <motion.div
   430→   430→                      className="h-full rounded-full"
   431→   431→                      style={{
   432→   432→                        background: isComplete
   433→   433→                          ? 'linear-gradient(90deg, #22C55E 0%, #4ADE80 100%)'
   434→   434→                          : 'linear-gradient(90deg, hsl(var(--primary)) 0%, #FFD60A 100%)',
   435→   435→                      }}
   436→   436→                      initial={{ width: 0 }}
   437→   437→                      animate={{ width: `${progress}%` }}
   438→   438→                      transition={{ duration: 0.6, delay: 0.2, ease: 'easeOut' }}
   439→   439→                    />
   440→   440→                  </div>
   441→   441→                </motion.div>
   442→   442→              )
   443→   443→            })}
   444→   444→          </motion.div>
   445→   445→        )}
   446→   446→      </motion.div>
   447→   447→
   448→   448→      {/* ─── Create List Dialog ─── */}
   449→   449→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   450→   450→        <DialogContent className="rounded-2xl border-border bg-background sm:max-w-md">
   451→   451→          <DialogHeader>
   452→   452→            <DialogTitle className="text-foreground">Yeni Liste Oluştur</DialogTitle>
   453→   453→            <DialogDescription className="text-muted-foreground">
   454→   454→              Alışveriş listenize bir isim verin
   455→   455→            </DialogDescription>
   456→   456→          </DialogHeader>
   457→   457→          <div className="flex flex-col gap-4 py-2">
   458→   458→            <Input
   459→   459→              value={newListName}
   460→   460→              onChange={(e) => setNewListName(titleCase(e.target.value))}
   461→   461→              onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   462→   462→              placeholder="Örn: Haftalık market alışverişi"
   463→   463→              className="border-primary/15 bg-background text-foreground placeholder:text-muted-foreground/50 focus-visible:border-primary focus-visible:ring-primary/30 glass-input"
   464→   464→              autoFocus
   465→   465→            />
   466→   466→          </div>
   467→   467→          <DialogFooter className="flex-row gap-2">
   468→   468→            <DialogClose asChild>
   469→   469→              <Button
   470→   470→                variant="ghost"
   471→   471→                className="flex-1 text-muted-foreground hover:bg-primary/5 hover:text-foreground"
   472→   472→              >
   473→   473→                İptal
   474→   474→              </Button>
   475→   475→            </DialogClose>
   476→   476→            <Button
   477→   477→              onClick={handleCreate}
   478→   478→              disabled={!newListName.trim() || creating}
   479→   479→              className="flex-1 rounded-full bg-primary text-primary-foreground hover:bg-primary/90"
   480→   480→            >
   481→   481→              {creating && <Loader2 className="mr-2 size-4 animate-spin" />}
   482→   482→              Oluştur
   483→   483→            </Button>
   484→   484→          </DialogFooter>
   485→   485→        </DialogContent>
   486→   486→      </Dialog>
   487→   487→
   488→   488→      {/* ─── Edit List Dialog ─── */}
   489→   489→      <Dialog open={editDialogOpen} onOpenChange={setEditDialogOpen}>
   490→   490→        <DialogContent className="rounded-2xl border-border bg-background sm:max-w-md">
