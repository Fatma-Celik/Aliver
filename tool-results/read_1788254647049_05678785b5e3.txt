   535→function ItemsView() {
   536→  const { token, activeList, setActiveList, setLists, lists } = useAppStore()
   537→  const [items, setItems] = useState<ShoppingItem[]>([])
   538→  const [loading, setLoading] = useState(true)

   546→  const [searchQuery, setSearchQuery] = useState('')
   547→  const inputRef = useRef<HTMLInputElement>(null)
   548→
   549→  /* ---------- Fetch items ---------- */
   550→  const fetchItems = useCallback(async () => {
   551→    if (!token || !activeList) return
   552→    setLoading(true)
   553→    try {
   554→      const res = await fetch(
   555→        `/api/lists/items?listId=${activeList.id}`,
   556→        { headers: authHeaders(token) },
   557→      )
   558→      if (res.ok) {
   559→        const data = await res.json()
   560→        setItems(data.items ?? [])
   561→      }
   562→    } catch {
   563→      // silent
   564→    } finally {
   565→      setLoading(false)
   566→    }
   567→  }, [token, activeList])
   568→
   569→  useEffect(() => {
   570→    fetchItems()
   571→  }, [fetchItems])
   572→
   573→  /* ---------- Computed ---------- */
   574→  const filteredItems = (activeList?.items ?? items).filter(item =>
   575→    item.name.toLowerCase().includes(searchQuery.toLowerCase())
   576→  )
   577→  const totalItems = items.length
   578→  const completedCount = items.filter((i) => i.completed).length
   579→  const progress = totalItems > 0 ? (completedCount / totalItems) * 100 : 0
   580→  const allCompleted = totalItems > 0 && completedCount === totalItems
   581→
   582→  const sortedItems = [...filteredItems].sort((a, b) => {
   583→    if (a.completed !== b.completed) return a.completed ? 1 : -1
   584→    return 0
   585→  })
   586→
   587→  /* ---------- Handlers ---------- */
   588→  const handleBack = () => {
   589→    setActiveList(null)
   590→  }
   591→
   592→  const handleAddItem = async () => {
   593→    if (!token || !activeList || !itemName.trim()) return
   594→    setAdding(true)
   595→    try {
   596→      const res = await fetch('/api/lists/items', {
   597→        method: 'POST',
   598→        headers: { 'Content-Type': 'application/json', ...authHeaders(token) },
   599→        body: JSON.stringify({
   600→          listId: activeList.id,
   601→          name: itemName.trim(),
   602→          quantity: Number(itemQty) || 1,
   603→          unit: itemUnit,
   604→        }),
   605→      })
   606→      if (res.ok) {
   607→        toast.success('Ürün eklendi')
   608→        setItemName('')
   609→        setItemQty('1')
   610→        setItemUnit('adet')
   611→        setAddItemOpen(false)
   612→        fetchItems()
   613→      } else {
   614→        toast.error('Ürün eklenemedi')
   615→      }
   616→    } catch {
   617→      toast.error('Bağlantı hatası')
   618→    } finally {
   619→      setAdding(false)
   620→    }
   621→  }
   622→
   623→  const handleToggle = async (itemId: string) => {
   624→    if (!token) return
   625→    setTogglingId(itemId)
   626→    try {
   627→      const res = await fetch('/api/lists/items', {
   628→        method: 'PATCH',
   629→        headers: { 'Content-Type': 'application/json', ...authHeaders(token) },
   630→        body: JSON.stringify({ itemId }),
   631→      })
   632→      if (res.ok) {
   633→        const data = await res.json()
   634→        setItems((prev) =>
   635→          prev.map((item) => (item.id === itemId ? { ...item, ...data.item } : item)),
   636→        )
   637→        // Update list counts in the store
   638→        setLists(
   639→          lists.map((l) => {
   640→            if (l.id === activeList?.id) {
   641→              const newCompleted = data.item.completed
   642→                ? (l._count?.completed ?? 0) + 1
   643→                : (l._count?.completed ?? 0) - 1
   644→              return { ...l, _count: { ...l._count!, completed: newCompleted } }
   645→            }
   646→            return l
   647→          }),
   648→        )
   649→      } else {
   650→        toast.error('Güncellenemedi')
   651→      }
   652→    } catch {
   653→      toast.error('Bağlantı hatası')
   654→    } finally {
   655→      setTogglingId(null)
   656→    }
   657→  }
   658→
   659→  const handleDelete = async (itemId: string) => {
   660→    if (!token) return
   661→    setDeletingId(itemId)
   662→    try {
   663→      const res = await fetch('/api/lists/items', {
   664→        method: 'DELETE',
   665→        headers: { 'Content-Type': 'application/json', ...authHeaders(token) },
   666→        body: JSON.stringify({ itemId }),
   667→      })
   668→      if (res.ok) {
   669→        toast.success('Ürün silindi')
   670→        setItems((prev) => prev.filter((item) => item.id !== itemId))
   671→        // Update list counts in the store
   672→        const deletedItem = items.find((i) => i.id === itemId)
   673→        setLists(
   674→          lists.map((l) => {
   675→            if (l.id === activeList?.id) {
   676→              return {
   677→                ...l,
   678→                _count: {
   679→                  items: (l._count?.items ?? 1) - 1,
   680→                  completed: (l._count?.completed ?? 0) - (deletedItem?.completed ? 1 : 0),
   681→                },
   682→              }
   683→            }
   684→            return l
   685→          }),
   686→        )
   687→      } else {
   688→        toast.error('Silinemedi')
   689→      }
   690→    } catch {
   691→      toast.error('Bağlantı hatası')
   692→    } finally {
   693→      setDeletingId(null)
   694→    }
   695→  }
   696→
   697→  /* ---------- Loading ---------- */
   698→  if (loading) {
   699→    return (
   700→      <div className="flex min-h-screen items-center justify-center bg-background">
   701→        <Loader2 className="size-8 animate-spin text-primary" />
   702→      </div>
   703→    )
   704→  }
   705→
   706→  return (
   707→    <main className="mx-auto min-h-screen w-full max-w-lg bg-background px-4 pb-24 pt-6">
   708→      <motion.div
   709→        variants={containerVariants}
   710→        initial="hidden"
   711→        animate="visible"
   712→        className="flex flex-col gap-5"
   713→      >
   714→        {/* ─── Header ─── */}
   715→        <motion.section
   716→          variants={itemVariants}
   717→          className="flex items-center gap-3"
   718→        >
   719→          <Button
   720→            onClick={handleBack}
   721→            variant="ghost"
   722→            size="icon"
   723→            className="shrink-0 text-muted-foreground hover:bg-primary/5 hover:text-foreground"
   724→          >
   725→            <ArrowLeft className="size-5" />
   726→          </Button>
   727→          <h1 className="min-w-0 truncate text-xl font-bold text-foreground">
   728→            {activeList?.name}
   729→          </h1>
   730→        </motion.section>
   731→
   732→        {/* ─── Search Bar ─── */}
   733→        {!(searchQuery === '' && items.length === 0) && (
   734→          <motion.div
   735→            variants={itemVariants}
   736→            className="flex items-center gap-2"
   737→          >
   738→            <div className="glass-card flex flex-1 items-center gap-2 rounded-full px-4 py-2">
   739→              <Search className="size-4 shrink-0 text-muted-foreground" />
   740→              <input
   741→                type="text"
   742→                value={searchQuery}
   743→                onChange={(e) => setSearchQuery(e.target.value)}
   744→                placeholder="Ürün ara..."
   745→                className="min-w-0 flex-1 bg-transparent text-sm text-foreground placeholder:text-muted-foreground/50 outline-none"
   746→              />
   747→              {searchQuery && (
   748→                <button
   749→                  onClick={() => setSearchQuery('')}
   750→                  className="shrink-0 text-muted-foreground/50 hover:text-muted-foreground"
   751→                >
   752→                  <AlertCircle className="size-3.5" />
   753→                </button>
   754→              )}
   755→            </div>
   756→            <span className="shrink-0 text-xs text-muted-foreground">
   757→              {filteredItems.length} ürün
   758→            </span>
   759→          </motion.div>
   760→        )}
   761→
   762→        {/* ─── Summary Bar ─── */}
   763→        <motion.section
   764→          variants={itemVariants}
   765→          className="glass-card rounded-2xl p-4"
   766→        >
   767→          <div className="flex items-center justify-between">
   768→            <div className="flex items-center gap-2">
   769→              <ShoppingCart className="size-4 text-primary" />
   770→              <span className="text-sm text-muted-foreground">
   771→                <span className="font-semibold text-primary">
   772→                  {completedCount}
   773→                </span>
   774→                {' / '}{totalItems} tamamlandı
   775→              </span>
   776→            </div>
   777→            <Badge
   778→              className="bg-primary/15 text-xs text-primary hover:bg-primary/20"
   779→            >
   780→              {totalItems > 0 ? Math.round(progress) : 0}%
   781→            </Badge>
   782→          </div>
   783→          <div className="mt-3 h-2 w-full overflow-hidden rounded-full bg-border">
   784→            <motion.div
   785→              className="h-full rounded-full"
   786→              style={{
   787→                background: allCompleted
   788→                  ? 'linear-gradient(90deg, #22C55E 0%, #4ADE80 100%)'
   789→                  : 'linear-gradient(90deg, hsl(var(--primary)) 0%, #FFD60A 100%)',
   790→              }}
   791→              initial={{ width: 0 }}
   792→              animate={{ width: `${progress}%` }}
   793→              transition={{ duration: 0.6, delay: 0.2, ease: 'easeOut' }}
   794→            />
   795→          </div>
   796→        </motion.section>
   797→
   798→        {/* ─── Celebration Message ─── */}
   799→        <AnimatePresence>
   800→          {allCompleted && (
   801→            <motion.div
   802→              initial={{ opacity: 0, scale: 0.9, y: -10 }}
   803→              animate={{ opacity: 1, scale: 1, y: 0 }}
   804→              exit={{ opacity: 0, scale: 0.9, y: -10 }}
   805→              transition={{ type: 'spring', stiffness: 300, damping: 20 }}
   806→              className="rounded-2xl bg-gradient-to-r from-primary/10 via-primary/20 to-primary/10 p-4 text-center"
   807→            >
   808→              <div className="flex items-center justify-center gap-2">
   809→                <Sparkles className="size-5 text-primary" />
   810→                <p className="text-base font-semibold text-primary">
   811→                  Tüm alışveriş tamamlandı! 🎉
   812→                </p>
   813→                <Sparkles className="size-5 text-primary" />
   814→              </div>
   815→            </motion.div>
   816→          )}
   817→        </AnimatePresence>
   818→
   819→        {/* ─── Add Item Section ─── */}
   820→        <motion.div variants={itemVariants} className="flex flex-col gap-2">
   821→          {!addItemOpen ? (
   822→            <Button
   823→              onClick={() => {
   824→                setAddItemOpen(true)
   825→                setTimeout(() => inputRef.current?.focus(), 100)
   826→              }}
   827→              className="w-full justify-center gap-2 rounded-2xl border border-dashed border-primary/30 bg-primary/10 text-primary hover:bg-primary/15"
   828→            >
   829→              <Plus className="size-4" />
   830→              Ekle
   831→            </Button>
   832→          ) : (
   833→            <motion.div
   834→              initial={{ opacity: 0, height: 0 }}
   835→              animate={{ opacity: 1, height: 'auto' }}
   836→              exit={{ opacity: 0, height: 0 }}
   837→              className="glass-card flex flex-col gap-3 rounded-2xl p-4"
   838→            >
   839→              <Input
   840→                ref={inputRef}
   841→                value={itemName}
   842→                onChange={(e) => setItemName(titleCase(e.target.value))}
   843→                onKeyDown={(e) => e.key === 'Enter' && handleAddItem()}
   844→                placeholder="Ürün adı"
   845→                className="border-primary/15 bg-background text-foreground placeholder:text-muted-foreground/50 focus-visible:border-primary focus-visible:ring-primary/30 glass-input"
   846→              />
   847→              <div className="flex gap-2">
   848→                <Input
   849→                  value={itemQty}
   850→                  onChange={(e) => setItemQty(e.target.value)}
   851→                  onKeyDown={(e) => e.key === 'Enter' && handleAddItem()}
   852→                  placeholder="1"
   853→                  type="number"
   854→                  min={1}
   855→                  className="w-20 shrink-0 border-primary/15 bg-background text-center text-foreground placeholder:text-muted-foreground/50 focus-visible:border-primary focus-visible:ring-primary/30 glass-input"
   856→                />
   857→                <Select value={itemUnit} onValueChange={setItemUnit}>
   858→                  <SelectTrigger
   859→                    className="w-full shrink-0 border-primary/15 bg-background text-foreground focus:ring-primary/30 focus:border-primary"
   860→                  >
   861→                    <SelectValue />
   862→                  </SelectTrigger>
   863→                  <SelectContent className="border-border bg-background">
   864→                    {UNITS.map((u) => (
   865→                      <SelectItem
   866→                        key={u}
   867→                        value={u}
   868→                        className="text-foreground focus:bg-primary/15 focus:text-primary"
   869→                      >
   870→                        {u.charAt(0).toUpperCase() + u.slice(1)}
   871→                      </SelectItem>
   872→                    ))}
   873→                  </SelectContent>
   874→                </Select>
   875→              </div>
   876→              <div className="flex gap-2">
   877→                <Button
   878→                  onClick={() => {
   879→                    setAddItemOpen(false)
   880→                    setItemName('')
   881→                    setItemQty('1')
   882→                    setItemUnit('adet')
   883→                  }}
   884→                  variant="ghost"
   885→                  className="flex-1 text-muted-foreground hover:bg-primary/5 hover:text-foreground"
   886→                >
   887→                  İptal
   888→                </Button>
   889→                <Button
   890→                  onClick={handleAddItem}
   891→                  disabled={!itemName.trim() || adding}
   892→                  className="flex-1 rounded-full bg-primary text-primary-foreground hover:bg-primary/90 hover-shine"
   893→                >
   894→                  {adding && <Loader2 className="mr-2 size-4 animate-spin" />}
   895→                  <Plus className="mr-1 size-4" />
   896→                  Ekle
   897→                </Button>
   898→              </div>
   899→            </motion.div>
   900→          )}
   901→        </motion.div>
   902→
   903→        <Separator className="bg-border" />
   904→
   905→        {/* ─── Items List ─── */}
   906→        {filteredItems.length === 0 && searchQuery === '' ? (
   907→          <motion.div
   908→            initial={{ opacity: 0, scale: 0.95 }}
   909→            animate={{ opacity: 1, scale: 1 }}
   910→            transition={{ duration: 0.4, delay: 0.15 }}
   911→            className="glass-card flex flex-col items-center justify-center gap-3 rounded-2xl px-6 py-12"
   912→          >
   913→            <div className="flex size-14 items-center justify-center rounded-full bg-primary/5">
   914→              <Package className="size-7 text-muted-foreground/30" />
   915→            </div>
   916→            <div className="flex flex-col items-center gap-1">
   917→              <p className="text-sm font-medium text-muted-foreground">
   918→                Liste boş
   919→              </p>
   920→              <p className="text-xs text-muted-foreground/50">
   921→                İlk ürününüzü ekleyerek alışverişe başlayın
   922→              </p>
   923→            </div>
   924→          </motion.div>
   925→        ) : filteredItems.length === 0 && searchQuery !== '' ? (
   926→          <motion.div
   927→            initial={{ opacity: 0, scale: 0.95 }}
   928→            animate={{ opacity: 1, scale: 1 }}
   929→            transition={{ duration: 0.4, delay: 0.15 }}
   930→            className="glass-card flex flex-col items-center justify-center gap-3 rounded-2xl px-6 py-12"
   931→          >
   932→            <div className="flex size-14 items-center justify-center rounded-full bg-primary/5">
   933→              <Search className="size-7 text-muted-foreground/30" />
   934→            </div>
   935→            <div className="flex flex-col items-center gap-1">
   936→              <p className="text-sm font-medium text-muted-foreground">
   937→                Sonuç bulunamadı
   938→              </p>
   939→              <p className="text-xs text-muted-foreground/50">
   940→                "{searchQuery}" ile eşleşen ürün yok
   941→              </p>
   942→            </div>
   943→          </motion.div>
   944→        ) : (
   945→          <motion.div
   946→            variants={containerVariants}
   947→            initial="hidden"
   948→            animate="visible"
   949→            className="flex flex-col gap-2"
   950→          >
   951→            <AnimatePresence mode="popLayout">
   952→              {sortedItems.map((item) => (
   953→                <motion.div
   954→                  key={item.id}
   955→                  variants={slideInVariants}
   956→                  initial="hidden"
   957→                  animate="visible"
   958→                  exit="exit"
   959→                  layout
   960→                  className={`group relative rounded-2xl ${item.completed ? 'opacity-50' : ''}`}
   961→                >
   962→                  <div className="glass-card hover-border-glow flex items-center gap-3 rounded-2xl px-4 py-3">
   963→                    {/* Checkbox */}
   964→                    <motion.div
   965→                      whileTap={{ scale: 0.85 }}
   966→                      className="shrink-0"
   967→                    >
   968→                      <Checkbox
   969→                        checked={item.completed}
   970→                        disabled={togglingId === item.id}
   971→                        onCheckedChange={() => handleToggle(item.id)}
   972→                        className="size-5 rounded-md border-border data-[state=checked]:border-primary data-[state=checked]:bg-primary data-[state=checked]:text-primary-foreground"
   973→                      />
   974→                    </motion.div>
   975→
   976→                    {/* Item info */}
   977→                    <div className="min-w-0 flex-1">
   978→                      <p
   979→                        className={`text-sm font-medium ${
   980→                          item.completed
   981→                            ? 'text-muted-foreground/70 line-through'
   982→                            : 'text-foreground'
   983→                        }`}
   984→                      >
   985→                        {item.name}
   986→                      </p>
   987→                      <div className="mt-1 flex flex-wrap items-center gap-2">
   988→                        <Badge
   989→                          variant="outline"
   990→                          className="border-primary/15 bg-primary/10 text-xs text-primary"
   991→                        >
   992→                          {item.quantity} {item.unit}
   993→                        </Badge>
   994→                        <span className="text-xs text-muted-foreground/70">
   995→                          {item.adder?.name}
   996→                        </span>
   997→                        {item.completed && item.purchaser?.name && (
   998→                          <span className="text-xs text-green-500/70">
   999→                            <Check className="mr-0.5 inline size-3" />
  1000→                            {item.purchaser.name}
  1001→                          </span>
  1002→                        )}
  1003→                      </div>
  1004→                    </div>
  1005→
  1006→                    {/* Delete button */}
  1007→                    <motion.div
  1008→                      initial={{ opacity: 0, scale: 0.8 }}
  1009→                      animate={{
  1010→                        opacity: deletingId === item.id ? 1 : 0.4,
  1011→                        scale: 1,
  1012→                      }}
  1013→                      whileHover={{ opacity: 1, scale: 1.1 }}
  1014→                      whileTap={{ scale: 0.9 }}
  1015→                      className="shrink-0"
  1016→                    >
  1017→                      <Button
  1018→                        variant="ghost"
  1019→                        size="icon"
  1020→                        className="size-8 text-destructive/60 hover:bg-destructive/10 hover:text-destructive"
  1021→                        onClick={(e) => {
  1022→                          e.stopPropagation()
  1023→                          handleDelete(item.id)
  1024→                        }}
  1025→                        disabled={deletingId === item.id}
  1026→                      >
  1027→                        {deletingId === item.id ? (
  1028→                          <Loader2 className="size-4 animate-spin" />
  1029→                        ) : (
  1030→                          <Trash2 className="size-4" />
  1031→                        )}
  1032→                      </Button>
  1033→                    </motion.div>
  1034→                  </div>
  1035→                </motion.div>
  1036→              ))}
  1037→            </AnimatePresence>
  1038→          </motion.div>
  1039→        )}
  1040→      </motion.div>
  1041→    </main>
  1042→  )
  1043→}
