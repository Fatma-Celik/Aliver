   449→function ItemsView() {
   450→  const { token, activeList, setActiveList, setLists, lists } = useAppStore()
   451→  const [items, setItems] = useState<ShoppingItem[]>([])
   452→  const [loading, setLoading] = useState(true)
   453→  const [addItemOpen, setAddItemOpen] = useState(false)
   454→  const [itemName, setItemName] = useState('')
   455→  const [itemQty, setItemQty] = useState('1')
   456→  const [itemUnit, setItemUnit] = useState('adet')
   457→  const [adding, setAdding] = useState(false)
   458→  const [togglingId, setTogglingId] = useState<string | null>(null)
   459→  const [deletingId, setDeletingId] = useState<string | null>(null)
   460→  const [searchQuery, setSearchQuery] = useState('')
   461→  const inputRef = useRef<HTMLInputElement>(null)
   462→
   463→  /* ---------- Fetch items ---------- */
   464→  const fetchItems = useCallback(async () => {
   465→    if (!token || !activeList) return
   466→    setLoading(true)
   467→    try {
   468→      const res = await fetch(
   469→        `/api/lists/items?listId=${activeList.id}`,
   470→        { headers: authHeaders(token) },
   471→      )
   472→      if (res.ok) {
   473→        const data = await res.json()
   474→        setItems(data.items ?? [])
   475→      }
   476→    } catch {
   477→      // silent
   478→    } finally {
   479→      setLoading(false)
   480→    }
   481→  }, [token, activeList])
   482→
   483→  useEffect(() => {
   484→    fetchItems()
   485→  }, [fetchItems])
   486→
   487→  /* ---------- Computed ---------- */
   488→  const filteredItems = (activeList?.items ?? items).filter(item =>
   489→    item.name.toLowerCase().includes(searchQuery.toLowerCase())
   490→  )
   491→  const totalItems = items.length
   492→  const completedCount = items.filter((i) => i.completed).length
   493→  const progress = totalItems > 0 ? (completedCount / totalItems) * 100 : 0
   494→  const allCompleted = totalItems > 0 && completedCount === totalItems
   495→
   496→  const sortedItems = [...filteredItems].sort((a, b) => {
   497→    if (a.completed !== b.completed) return a.completed ? 1 : -1
   498→    return 0
   499→  })
   500→
   501→  /* ---------- Handlers ---------- */
   502→  const handleBack = () => {
   503→    setActiveList(null)
   504→  }
   505→
   506→  const handleAddItem = async () => {
   507→    if (!token || !activeList || !itemName.trim()) return
   508→    setAdding(true)
   509→    try {
   510→      const res = await fetch('/api/lists/items', {
   511→        method: 'POST',
   512→        headers: { 'Content-Type': 'application/json', ...authHeaders(token) },
   513→        body: JSON.stringify({
   514→          listId: activeList.id,
   515→          name: itemName.trim(),
   516→          quantity: Number(itemQty) || 1,
   517→          unit: itemUnit,
   518→        }),
   519→      })
   520→      if (res.ok) {
   521→        toast.success('Ürün eklendi')
   522→        setItemName('')
   523→        setItemQty('1')
   524→        setItemUnit('adet')
   525→        setAddItemOpen(false)
   526→        fetchItems()
   527→      } else {
   528→        toast.error('Ürün eklenemedi')
   529→      }
   530→    } catch {
   531→      toast.error('Bağlantı hatası')
   532→    } finally {
   533→      setAdding(false)
   534→    }
   535→  }
   536→
   537→  const handleToggle = async (itemId: string) => {
   538→    if (!token) return
   539→    setTogglingId(itemId)
   540→    try {
   541→      const res = await fetch('/api/lists/items', {
   542→        method: 'PATCH',
   543→        headers: { 'Content-Type': 'application/json', ...authHeaders(token) },
   544→        body: JSON.stringify({ itemId }),
   545→      })
   546→      if (res.ok) {
   547→        const data = await res.json()
   548→        setItems((prev) =>
   549→          prev.map((item) => (item.id === itemId ? { ...item, ...data.item } : item)),
   550→        )
   551→        // Update list counts in the store
   552→        setLists(
   553→          lists.map((l) => {
   554→            if (l.id === activeList?.id) {
   555→              const newCompleted = data.item.completed
   556→                ? (l._count?.completed ?? 0) + 1
   557→                : (l._count?.completed ?? 0) - 1
   558→              return { ...l, _count: { ...l._count!, completed: newCompleted } }
   559→            }
   560→            return l
   561→          }),
   562→        )
   563→      } else {
   564→        toast.error('Güncellenemedi')
   565→      }
   566→    } catch {
   567→      toast.error('Bağlantı hatası')
   568→    } finally {
   569→      setTogglingId(null)
   570→    }
   571→  }
   572→
   573→  const handleDelete = async (itemId: string) => {
   574→    if (!token) return
   575→    setDeletingId(itemId)
   576→    try {
   577→      const res = await fetch('/api/lists/items', {
   578→        method: 'DELETE',
   579→        headers: { 'Content-Type': 'application/json', ...authHeaders(token) },
   580→        body: JSON.stringify({ itemId }),
   581→      })
   582→      if (res.ok) {
   583→        toast.success('Ürün silindi')
   584→        setItems((prev) => prev.filter((item) => item.id !== itemId))
   585→        // Update list counts in the store
   586→        const deletedItem = items.find((i) => i.id === itemId)
   587→        setLists(
   588→          lists.map((l) => {
   589→            if (l.id === activeList?.id) {
   590→              return {
   591→                ...l,
   592→                _count: {
   593→                  items: (l._count?.items ?? 1) - 1,
   594→                  completed: (l._count?.completed ?? 0) - (deletedItem?.completed ? 1 : 0),
   595→                },
   596→              }
   597→            }
   598→            return l
   599→          }),
   600→        )
   601→      } else {
   602→        toast.error('Silinemedi')
   603→      }
   604→    } catch {
   605→      toast.error('Bağlantı hatası')
   606→    } finally {
   607→      setDeletingId(null)
   608→    }
   609→  }
   610→
   611→  /* ---------- Loading ---------- */
   612→  if (loading) {
   613→    return (
   614→      <div className="flex min-h-screen items-center justify-center bg-background">
   615→        <Loader2 className="size-8 animate-spin text-primary" />
   616→      </div>
   617→    )
   618→  }
   619→
   620→  return (
   621→    <main className="mx-auto min-h-screen w-full max-w-lg bg-background px-4 pb-24 pt-6">
   622→      <motion.div
   623→        variants={containerVariants}
   624→        initial="hidden"
   625→        animate="visible"
   626→        className="flex flex-col gap-5"
   627→      >
   628→        {/* ─── Header ─── */}
   629→        <motion.section
   630→          variants={itemVariants}
   631→          className="flex items-center gap-3"
   632→        >
   633→          <Button
   634→            onClick={handleBack}
   635→            variant="ghost"
   636→            size="icon"
   637→            className="shrink-0 text-muted-foreground hover:bg-primary/5 hover:text-foreground"
   638→          >
   639→            <ArrowLeft className="size-5" />
   640→          </Button>
   641→          <h1 className="min-w-0 truncate text-xl font-bold text-foreground">
   642→            {activeList?.name}
   643→          </h1>
   644→        </motion.section>
   645→
   646→        {/* ─── Search Bar ─── */}
   647→        {!(searchQuery === '' && items.length === 0) && (
   648→          <motion.div
   649→            variants={itemVariants}
   650→            className="flex items-center gap-2"
   651→          >
   652→            <div className="glass-card flex flex-1 items-center gap-2 rounded-full px-4 py-2">
   653→              <Search className="size-4 shrink-0 text-muted-foreground" />
   654→              <input
   655→                type="text"
   656→                value={searchQuery}
   657→                onChange={(e) => setSearchQuery(e.target.value)}
   658→                placeholder="Ürün ara..."
   659→                className="min-w-0 flex-1 bg-transparent text-sm text-foreground placeholder:text-muted-foreground/50 outline-none"
   660→              />
   661→              {searchQuery && (
   662→                <button
   663→                  onClick={() => setSearchQuery('')}
   664→                  className="shrink-0 text-muted-foreground/50 hover:text-muted-foreground"
   665→                >
   666→                  <AlertCircle className="size-3.5" />
   667→                </button>
   668→              )}
   669→            </div>
   670→            <span className="shrink-0 text-xs text-muted-foreground">
   671→              {filteredItems.length} ürün
   672→            </span>
   673→          </motion.div>
   674→        )}
   675→
   676→        {/* ─── Summary Bar ─── */}
   677→        <motion.section
   678→          variants={itemVariants}
   679→          className="glass-card rounded-2xl p-4"
   680→        >
   681→          <div className="flex items-center justify-between">
   682→            <div className="flex items-center gap-2">
   683→              <ShoppingCart className="size-4 text-primary" />
   684→              <span className="text-sm text-muted-foreground">
   685→                <span className="font-semibold text-primary">
   686→                  {completedCount}
   687→                </span>
   688→                {' / '}{totalItems} tamamlandı
   689→              </span>
   690→            </div>
   691→            <Badge
   692→              className="bg-primary/15 text-xs text-primary hover:bg-primary/20"
   693→            >
   694→              {totalItems > 0 ? Math.round(progress) : 0}%
   695→            </Badge>
   696→          </div>
   697→          <div className="mt-3 h-2 w-full overflow-hidden rounded-full bg-border">
   698→            <motion.div
   699→              className="h-full rounded-full"
   700→              style={{
   701→                background: allCompleted
   702→                  ? 'linear-gradient(90deg, #22C55E 0%, #4ADE80 100%)'
   703→                  : 'linear-gradient(90deg, hsl(var(--primary)) 0%, #FFD60A 100%)',
   704→              }}
   705→              initial={{ width: 0 }}
   706→              animate={{ width: `${progress}%` }}
   707→              transition={{ duration: 0.6, delay: 0.2, ease: 'easeOut' }}
   708→            />
   709→          </div>
   710→        </motion.section>
   711→
   712→        {/* ─── Celebration Message ─── */}
   713→        <AnimatePresence>
   714→          {allCompleted && (
   715→            <motion.div
   716→              initial={{ opacity: 0, scale: 0.9, y: -10 }}
   717→              animate={{ opacity: 1, scale: 1, y: 0 }}
   718→              exit={{ opacity: 0, scale: 0.9, y: -10 }}
   719→              transition={{ type: 'spring', stiffness: 300, damping: 20 }}
   720→              className="rounded-2xl bg-gradient-to-r from-primary/10 via-primary/20 to-primary/10 p-4 text-center"
   721→            >
   722→              <div className="flex items-center justify-center gap-2">
   723→                <Sparkles className="size-5 text-primary" />
   724→                <p className="text-base font-semibold text-primary">
   725→                  Tüm alışveriş tamamlandı! 🎉
   726→                </p>
   727→                <Sparkles className="size-5 text-primary" />
   728→              </div>
   729→            </motion.div>
   730→          )}
   731→        </AnimatePresence>
   732→
   733→        {/* ─── Add Item Section ─── */}
   734→        <motion.div variants={itemVariants} className="flex flex-col gap-2">
   735→          {!addItemOpen ? (
   736→            <Button
   737→              onClick={() => {
   738→                setAddItemOpen(true)
   739→                setTimeout(() => inputRef.current?.focus(), 100)
   740→              }}
   741→              className="w-full justify-center gap-2 rounded-2xl border border-dashed border-primary/30 bg-primary/10 text-primary hover:bg-primary/15"
   742→            >
   743→              <Plus className="size-4" />
   744→              Ekle
   745→            </Button>
   746→          ) : (
   747→            <motion.div
   748→              initial={{ opacity: 0, height: 0 }}
   749→              animate={{ opacity: 1, height: 'auto' }}
   750→              exit={{ opacity: 0, height: 0 }}
   751→              className="glass-card flex flex-col gap-3 rounded-2xl p-4"
   752→            >
   753→              <Input
   754→                ref={inputRef}
   755→                value={itemName}
   756→                onChange={(e) => setItemName(titleCase(e.target.value))}
   757→                onKeyDown={(e) => e.key === 'Enter' && handleAddItem()}
   758→                placeholder="Ürün adı"
   759→                className="border-primary/15 bg-background text-foreground placeholder:text-muted-foreground/50 focus-visible:border-primary focus-visible:ring-primary/30 glass-input"
   760→              />
   761→              <div className="flex gap-2">
   762→                <Input
   763→                  value={itemQty}
   764→                  onChange={(e) => setItemQty(e.target.value)}
   765→                  onKeyDown={(e) => e.key === 'Enter' && handleAddItem()}
   766→                  placeholder="1"
   767→                  type="number"
   768→                  min={1}
   769→                  className="w-20 shrink-0 border-primary/15 bg-background text-center text-foreground placeholder:text-muted-foreground/50 focus-visible:border-primary focus-visible:ring-primary/30 glass-input"
   770→                />
   771→                <Select value={itemUnit} onValueChange={setItemUnit}>
   772→                  <SelectTrigger
   773→                    className="w-full shrink-0 border-primary/15 bg-background text-foreground focus:ring-primary/30 focus:border-primary"
   774→                  >
   775→                    <SelectValue />
   776→                  </SelectTrigger>
   777→                  <SelectContent className="border-border bg-background">
   778→                    {UNITS.map((u) => (
   779→                      <SelectItem
   780→                        key={u}
   781→                        value={u}
   782→                        className="text-foreground focus:bg-primary/15 focus:text-primary"
   783→                      >
   784→                        {u.charAt(0).toUpperCase() + u.slice(1)}
   785→                      </SelectItem>
   786→                    ))}
   787→                  </SelectContent>
   788→                </Select>
   789→              </div>
   790→              <div className="flex gap-2">
   791→                <Button
   792→                  onClick={() => {
   793→                    setAddItemOpen(false)
   794→                    setItemName('')
   795→                    setItemQty('1')
   796→                    setItemUnit('adet')
   797→                  }}
   798→                  variant="ghost"
   799→                  className="flex-1 text-muted-foreground hover:bg-primary/5 hover:text-foreground"
   800→                >
   801→                  İptal
   802→                </Button>
   803→                <Button
   804→                  onClick={handleAddItem}
   805→                  disabled={!itemName.trim() || adding}
   806→                  className="flex-1 rounded-full bg-primary text-primary-foreground hover:bg-primary/90 hover-shine"
   807→                >
   808→                  {adding && <Loader2 className="mr-2 size-4 animate-spin" />}
   809→                  <Plus className="mr-1 size-4" />
   810→                  Ekle
   811→                </Button>
   812→              </div>
   813→            </motion.div>
   814→          )}
   815→        </motion.div>
   816→
   817→        <Separator className="bg-border" />
   818→
   819→        {/* ─── Items List ─── */}
   820→        {filteredItems.length === 0 && searchQuery === '' ? (
   821→          <motion.div
   822→            initial={{ opacity: 0, scale: 0.95 }}
   823→            animate={{ opacity: 1, scale: 1 }}
   824→            transition={{ duration: 0.4, delay: 0.15 }}
   825→            className="glass-card flex flex-col items-center justify-center gap-3 rounded-2xl px-6 py-12"
   826→          >
   827→            <div className="flex size-14 items-center justify-center rounded-full bg-primary/5">
   828→              <Package className="size-7 text-muted-foreground/30" />
   829→            </div>
   830→            <div className="flex flex-col items-center gap-1">
   831→              <p className="text-sm font-medium text-muted-foreground">
   832→                Liste boş
   833→              </p>
   834→              <p className="text-xs text-muted-foreground/50">
   835→                İlk ürününüzü ekleyerek alışverişe başlayın
   836→              </p>
   837→            </div>
   838→          </motion.div>
   839→        ) : filteredItems.length === 0 && searchQuery !== '' ? (
   840→          <motion.div
   841→            initial={{ opacity: 0, scale: 0.95 }}
   842→            animate={{ opacity: 1, scale: 1 }}
   843→            transition={{ duration: 0.4, delay: 0.15 }}
   844→            className="glass-card flex flex-col items-center justify-center gap-3 rounded-2xl px-6 py-12"
   845→          >
   846→            <div className="flex size-14 items-center justify-center rounded-full bg-primary/5">
   847→              <Search className="size-7 text-muted-foreground/30" />
   848→            </div>
   849→            <div className="flex flex-col items-center gap-1">
   850→              <p className="text-sm font-medium text-muted-foreground">
   851→                Sonuç bulunamadı
   852→              </p>
   853→              <p className="text-xs text-muted-foreground/50">
   854→                "{searchQuery}" ile eşleşen ürün yok
   855→              </p>
   856→            </div>
   857→          </motion.div>
   858→        ) : (
   859→          <motion.div
   860→            variants={containerVariants}
   861→            initial="hidden"
   862→            animate="visible"
   863→            className="flex flex-col gap-2"
   864→          >
   865→            <AnimatePresence mode="popLayout">
   866→              {sortedItems.map((item) => (
   867→                <motion.div
   868→                  key={item.id}
   869→                  variants={slideInVariants}
   870→                  initial="hidden"
   871→                  animate="visible"
   872→                  exit="exit"
   873→                  layout
   874→                  className={`group relative rounded-2xl ${item.completed ? 'opacity-50' : ''}`}
   875→                >
   876→                  <div className="glass-card hover-border-glow flex items-center gap-3 rounded-2xl px-4 py-3">
   877→                    {/* Checkbox */}
   878→                    <motion.div
   879→                      whileTap={{ scale: 0.85 }}
   880→                      className="shrink-0"
   881→                    >
   882→                      <Checkbox
   883→                        checked={item.completed}
   884→                        disabled={togglingId === item.id}
   885→                        onCheckedChange={() => handleToggle(item.id)}
   886→                        className="size-5 rounded-md border-border data-[state=checked]:border-primary data-[state=checked]:bg-primary data-[state=checked]:text-primary-foreground"
   887→                      />
   888→                    </motion.div>
   889→
   890→                    {/* Item info */}
   891→                    <div className="min-w-0 flex-1">
   892→                      <p
   893→                        className={`text-sm font-medium ${
   894→                          item.completed
   895→                            ? 'text-muted-foreground/70 line-through'
   896→                            : 'text-foreground'
   897→                        }`}
   898→                      >
   899→                        {item.name}
   900→                      </p>
   901→                      <div className="mt-1 flex flex-wrap items-center gap-2">
   902→                        <Badge
   903→                          variant="outline"
   904→                          className="border-primary/15 bg-primary/10 text-xs text-primary"
   905→                        >
   906→                          {item.quantity} {item.unit}
   907→                        </Badge>
   908→                        <span className="text-xs text-muted-foreground/70">
   909→                          {item.adder?.name}
   910→                        </span>
   911→                        {item.completed && item.purchaser?.name && (
   912→                          <span className="text-xs text-green-500/70">
   913→                            <Check className="mr-0.5 inline size-3" />
   914→                            {item.purchaser.name}
   915→                          </span>
   916→                        )}
   917→                      </div>
   918→                    </div>
   919→
   920→                    {/* Delete button */}
   921→                    <motion.div
   922→                      initial={{ opacity: 0, scale: 0.8 }}
   923→                      animate={{
   924→                        opacity: deletingId === item.id ? 1 : 0.4,
   925→                        scale: 1,
   926→                      }}
   927→                      whileHover={{ opacity: 1, scale: 1.1 }}
   928→                      whileTap={{ scale: 0.9 }}
   929→                      className="shrink-0"
   930→                    >
   931→                      <Button
   932→                        variant="ghost"
   933→                        size="icon"
   934→                        className="size-8 text-destructive/60 hover:bg-destructive/10 hover:text-destructive"
   935→                        onClick={(e) => {
   936→                          e.stopPropagation()
   937→                          handleDelete(item.id)
   938→                        }}
   939→                        disabled={deletingId === item.id}
   940→                      >
   941→                        {deletingId === item.id ? (
   942→                          <Loader2 className="size-4 animate-spin" />
   943→                        ) : (
   944→                          <Trash2 className="size-4" />
   945→                        )}
   946→                      </Button>
   947→                    </motion.div>
   948→                  </div>
   949→                </motion.div>
   950→              ))}
   951→            </AnimatePresence>
   952→          </motion.div>
   953→        )}
   954→      </motion.div>
   955→    </main>
   956→  )
   957→}
